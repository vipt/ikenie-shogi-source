cmake_minimum_required(VERSION 3.22)
project(meguru LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(sources.cmake)
add_library(meguru_core OBJECT ${MEGURU_ENGINE_SOURCES} meguru_mobile.cpp)
set_target_properties(meguru_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(meguru_core PRIVATE YANEURAOU_ENGINE_NNUE NDEBUG IS_64BIT ENGINE_NAME="MeguruShogi-Mobile-1" TARGET_CPU="MOBILE")
target_compile_options(meguru_core PRIVATE -O3 -include "${CMAKE_CURRENT_SOURCE_DIR}/mobile_platform.h")
if(ANDROID)
  add_library(meguru SHARED meguru_jni.cpp $<TARGET_OBJECTS:meguru_core>)
  target_link_libraries(meguru log android)
endif()
if(MEGURU_CLI)
  add_executable(meguru_cli meguru_cli.cpp $<TARGET_OBJECTS:meguru_core>)
  if(NOT ANDROID)
    target_link_libraries(meguru_cli pthread)
  endif()
endif()

# Offline book research uses the exact mobile rules with configurable USI MultiPV.
if(MEGURU_CLI)
  add_executable(meguru_book_research meguru_book_research.cpp $<TARGET_OBJECTS:meguru_core>)
  target_compile_definitions(meguru_book_research PRIVATE YANEURAOU_ENGINE_NNUE NDEBUG IS_64BIT)
  target_compile_options(meguru_book_research PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/mobile_platform.h")
endif()
